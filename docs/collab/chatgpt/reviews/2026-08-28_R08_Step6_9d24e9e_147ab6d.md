## Accepted parts

\`\`\`python
present = history_mask.bool().any(dim=1)
data_batch["local_memory"] = [
    tokens[i, 0] if bool(present[i]) else None
    for i in range(tokens.shape[0])

for plan, has_local in zip(sequence_plans, present.tolist(), strict=True):
