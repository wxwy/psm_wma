"""GPU paired gradient parity harness: Compact=0 (Dense) vs Compact=1.

4-rank FSDP2 native model, single GA member (plan.members[0]), zero optimizer
step, zero checkpoint write. Same model instance / frozen DROID weights, same
frozen CPU plan, same per-rank RNG restored before each side.

Run (owner-authorized GPU4-7 coresidency, psm_wma venv):
  CUDA_VISIBLE_DEVICES=4,5,6,7 PYTHONPATH=<child> HF_HUB_OFFLINE=1   
  TRANSFORMERS_OFFLINE=1 <psm_wma python> -m torch.distributed.run   
  --standalone --nproc_per_node=4 /tmp/v3_nott_compact_gpu_parity.py
"""

import argparse

import os

import time
from pathlib import Path

SRC = "/mnt/data/shenzhen/szrobot/logs/.tmp_backup/psm_wma_v3/cosmos-framework"
if SRC not in sys.path:
    sys.path.insert(0, SRC)

GENERATION_KEYS = (
    "moe_gen",
    "time_embedder",
    "vae2llm",
    "llm2vae",
    "action2llm",
    "llm2action",
    "action_modality_embed",
)

def _rng_snapshot():
    import numpy as np
    import torch

    return {
        "python": random.getstate(),
        "numpy": np.random.get_state(),
        "torch_cpu": torch.random.get_rng_state().clone(),
        "torch_cuda": [torch.cuda.get_rng_state(d).clone() for d in range(torch.cuda.device_count())],
    }

def _rng_restore(state):
    import numpy as np
    import torch

    random.setstate(state["python"])
    np.random.set_state(state["numpy"])
    torch.random.set_rng_state(state["torch_cpu"])
    for d, s in enumerate(state["torch_cuda"]):
        torch.cuda.set_rng_state(s, d)

def _local(grad):
    from torch.distributed.tensor import DTensor

    if isinstance(grad, DTensor):
        return grad.to_local()
    return grad

def _group_of(name):
    for key in GENERATION_KEYS:
        if name.startswith(key) or key in name:
            return key
    return None

def main() -> None:
    import torch
    import numpy as np
    import torch.distributed as dist
    import torch.distributed.checkpoint as dcp
    from torch.distributed.checkpoint import FileSystemReader

    from cosmos_framework.utils import distributed, misc
    from cosmos_framework.utils.lazy_config import instantiate
    from cosmos_framework.utils.context_managers import model_init
    from cosmos_framework.data.generator.action.datasets.robocasa_verified_index import (
        VerifiedExactWindowIndex,
    )
    from cosmos_framework.data.generator.action.datasets.robocasa_exact_window_cache import (
        RoboCasaExactWindowCacheCatalog,
    )
    from cosmos_framework.data.generator.action.datasets.robocasa_exact_window_policy import (
        CorrectedRoboCasaPolicyContract,
    )
    from cosmos_framework.data.generator.action.datasets.robocasa_exact_window_cached_sft import (
        get_action_robocasa_exact_window_cached_sft_dataset,
    )
    from cosmos_framework.data.generator.action.utils.cached_pixel_geometry import (
        move_cached_native_batch,
    )
    from cosmos_framework.model.generator.mot.robocasa_exact_window_local import (
        ExactWindowLocalCatalog,
        ExactWindowRankPlanner,
        ExactWindowSegmentProducer,
    )
    from cosmos_framework.trainer.matched_no_ttt import MatchedNoTTTWindow
    from cosmos_framework.trainer.local_memory_grouped import collate_grouped_native_batch
    from cosmos_framework.checkpoint.dcp import CustomLoadPlanner, ModelWrapper
    from examples import psm_wma_robocasa_corrected_phase5 as phase5

    phase5._load_runtime()

    parser = argparse.ArgumentParser()
    for name in (
        "source-root",
        "cache-root",
        "dataset-index-root",
        "edge",
        "vae",
        "base-checkpoint",
        "output-root",
    ):
        parser.add_argument(f"--{name}", required=True, type=Path)
    args = parser.parse_args()

    rank = int(os.environ["RANK"])
    local_rank = int(os.environ["LOCAL_RANK"])
    world_size = int(os.environ["WORLD_SIZE"])
    if world_size != 4:
        raise RuntimeError(f"harness requires world_size=4, got {world_size}")

    result = {
        "rank": rank,
        "local_rank": local_rank,
        "world_size": world_size,
        "torch": torch.**version**,
        "cuda": torch.version.cuda,
        "device": torch.cuda.get_device_name(local_rank),
        "started_unix": time.time(),
    }

    torch.manual_seed(0)
    random.seed(0)
    np.random.seed(0)
    distributed.init()
    dist.barrier()

    result["compact_source_roots"] = {
        "index": str(args.dataset_index_root),
        "cache": str(args.cache_root),
        "source": str(args.source_root),
    }

    vi = VerifiedExactWindowIndex.open(
        args.dataset_index_root,
        cache_root=args.cache_root,
        source_root=args.source_root,
        verify_files=False,
    )
    cache = RoboCasaExactWindowCacheCatalog(args.cache_root, verified_index=vi)
    contract = CorrectedRoboCasaPolicyContract.from_cache_catalog(cache)

    train_args = argparse.Namespace(
        no_ttt=True,
        t=16,
        b=8,
        ga=4,
        k=4,
        world_size=4,
        num_workers=0,
        max_iter=30000,
        warmup=500,
        save_iter=100,
        job_name="no_ttt_gpu_parity_harness",
        edge=args.edge,
        vae=args.vae,
        source_root=args.source_root,
        base_checkpoint=args.base_checkpoint,
        output_root=args.output_root,
        transform_rng_mode="legacy_global",
        full_segment_prefetch=False,
    )
    config = phase5._load_config(train_args)
    phase5.overlay_config(config, train_args, contract, resume_checkpoint=None)
    tokenizer = config.model.config.vlm_config.tokenizer

    def build_catalog(mode):
        os.environ["PSM_V3_COMPACT_CACHED_VIDEO"] = str(mode)
        dataset = get_action_robocasa_exact_window_cached_sft_dataset(
            cache_root=args.cache_root,
            source_root=args.source_root,
            tokenizer_config=tokenizer,
            catalog=cache,
            verified_index=vi,
        )
        catalog = ExactWindowLocalCatalog(dataset, ttt_tbptt_steps=16)
        os.environ.pop("PSM_V3_COMPACT_CACHED_VIDEO", None)
        return catalog

    catalog0 = build_catalog(0)
    catalog1 = build_catalog(1)
    result["compact0_catalog"] = {
        "episodes": len(catalog0.episodes),
        "compact_flag": catalog0.raw.compact_cached_video,
    }
    result["compact1_catalog"] = {
        "episodes": len(catalog1.episodes),
        "compact_flag": catalog1.raw.compact_cached_video,
    }

    planner0 = ExactWindowRankPlanner(
        catalog0, rank=rank, world_size=4, b_stream=8, active_ga=4, seed=0
    )
    planner1 = ExactWindowRankPlanner(
        catalog1, rank=rank, world_size=4, b_stream=8, active_ga=4, seed=0
    )
    random.seed(0)
    plan0 = planner0.plan_window(planner0.initial_frontier())
    random.seed(0)
    plan1 = planner1.plan_window(planner1.initial_frontier())

    members_eq = plan0.members[0] == plan1.members[0]
    n_window0 = int(plan0.n_window)
    n_window1 = int(plan1.n_window)
    member_counts0 = list(plan0.member_counts)
    result["plan"] = {
        "members_eq": bool(members_eq),
        "n_window0": n_window0,
        "n_window1": n_window1,
        "member_counts0": member_counts0,
        "member0_count": len(plan0.members[0]),
        "member1_count": len(plan1.members[0]),
    }
    if not members_eq or n_window0 != n_window1 or member_counts0[0] != 128:
        result["verdict"] = "FAIL_SOURCE_PLAN_MISMATCH"
        _emit(result, rank)
        return

    producer0 = ExactWindowSegmentProducer(catalog0, config_digest="gpu_parity_harness_v1")
    producer1 = ExactWindowSegmentProducer(catalog1, config_digest="gpu_parity_harness_v1")

    with model_init():
        model = instantiate(config.model)
    model = model.to("cuda", memory_format=torch.preserve_format)
    model.on_train_start()
    dist.barrier()

    dcp_state = ModelWrapper(model).state_dict()
    dcp.load(
        state_dict=dcp_state,
        storage_reader=FileSystemReader(str(args.base_checkpoint / "model")),
        planner=CustomLoadPlanner(keys_to_skip_loading=["net_ema.", "local_memory"]),
        no_dist=True,
    )
    load_result = ModelWrapper(model).load_state_dict(dcp_state)
    result["dcp_load"] = {
        "path": str(args.base_checkpoint),
        "missing_keys": list(load_result.missing_keys)[:20],
        "unexpected_keys": list(load_result.unexpected_keys)[:20],
        "missing_count": len(load_result.missing_keys),
        "unexpected_count": len(load_result.unexpected_keys),
    }
    del dcp_state
    dist.barrier()
    torch.cuda.synchronize()

    selected_names = [
        name
        for name, _ in model.named_parameters()
        if any(name.startswith(key) or key in name for key in GENERATION_KEYS)
    ]
    result["selected_params"] = {
        "count": len(selected_names),
        "groups": {key: 0 for key in GENERATION_KEYS},
    }
    for name in selected_names:
        g = _group_of(name)
        if g is not None:
            result["selected_params"]["groups"][g] += 1

    def run_side(producer, planner, label):
        window = MatchedNoTTTWindow(model, planner)
        window.begin()
        requests = window.plan.members[0]
        segments = tuple(producer.produce(req) for req in requests)
        payloads, positions = window.pack(segments)
        batch = collate_grouped_native_batch(payloads)
        batch = move_cached_native_batch(
            batch, device=next(model.parameters()).device, move=misc.to
        )
        output, loss = model.training_step(batch, 1000)
        weighted = loss * (len(payloads) / window.plan.n_window)
        weighted.backward()
        model.on_after_backward()
        torch.cuda.synchronize()
        dist.barrier()
        loss_value = float(loss.detach().cpu())
        weighted_value = float(weighted.detach().cpu())
        grads = {}
        for name, param in model.named_parameters():
            if param.grad is not None:
                grads[name] = _local(param.grad.detach()).clone()
        result[label] = {
            "loss": loss_value,
            "weighted": weighted_value,
            "n_payloads": len(payloads),
            "grad_params": len(grads),
        }
        print(
            f"[{label}] rank={rank} loss={loss_value:.9f} weighted={weighted_value:.9f} "
            f"n={len(payloads)} grads={len(grads)}",
            flush=True,
        )
        return grads

    rng0 = _rng_snapshot()

    grads_dense0a = run_side(producer0, planner0, "dense0a")

    for param in model.parameters():
        param.grad = None

    _rng_restore(rng0)
    grads_dense0b = run_side(producer0, planner0, "dense0b")

    for param in model.parameters():
        param.grad = None

    _rng_restore(rng0)
    grads_compact1 = run_side(producer1, planner1, "compact1")

    def compare(g0, g1, loss_a, loss_b, tag):
        names = sorted(set(g0) & set(g1))
        only_a = sorted(set(g0) - set(g1))
        only_b = sorted(set(g1) - set(g0))
        loss_eq = math_isclose(loss_a, loss_b, rel_tol=1e-4, abs_tol=1e-5)
        per_param = {}
        failures = []
        for name in names:
            ga = g0[name].float()
            gb = g1[name].float()
            if ga.shape != gb.shape:
                failures.append(name)
                continue
            max_abs = float((ga - gb).abs().max())
            denom = float(ga.norm())
            rel_l2 = 0.0 if denom == 0.0 else float((ga - gb).norm() / denom)
            ok = bool(torch.allclose(ga, gb, rtol=1e-3, atol=1e-5))
            per_param[name] = {"max_abs": max_abs, "rel_l2": rel_l2, "allclose": ok}
            if not ok:
                failures.append(name)
        for name in only_a + only_b:
            failures.append(f"grad_missing_on_one_side:{name}")
        group_stats = {key: {"covered": 0, "pass": 0} for key in GENERATION_KEYS}
        for name in names:
            g = _group_of(name)
            if g is None:
                continue
            group_stats[g]["covered"] += 1
            if per_param[name]["allclose"]:
                group_stats[g]["pass"] += 1
        worst = sorted(per_param.items(), key=lambda kv: -kv[1]["max_abs"])[:10]
        result[tag] = {
            "common_params": len(names),
            "only_a": only_a[:20],
            "only_b": only_b[:20],
            "failures": failures[:50],
            "failure_count": len(failures),
            "loss_eq": loss_eq,
            "loss_a": loss_a,
            "loss_b": loss_b,
            "group_stats": group_stats,
            "worst10": [
                {"name": name, "max_abs": v["max_abs"], "rel_l2": v["rel_l2"], "allclose": v["allclose"]}
                for name, v in worst
            ],
        }
        return per_param, names

    per_noise, names_noise = compare(
        grads_dense0a, grads_dense0b, result["dense0a"]["loss"], result["dense0b"]["loss"], "intrinsic_noise"
    )
    per_compact, names_compact = compare(
        grads_dense0a, grads_compact1, result["dense0a"]["loss"], result["compact1"]["loss"], "compact_effect"
    )

    common = sorted(set(per_noise) & set(per_compact))
    excess = {}
    for name in common:
        d_noise = per_noise[name]["max_abs"]
        d_compact = per_compact[name]["max_abs"]
        excess[name] = {
            "d_noise": d_noise,
            "d_compact": d_compact,
            "excess": d_compact - d_noise,
            "ratio": (d_compact / d_noise) if d_noise > 0.0 else None,
            "noise_ok": per_noise[name]["allclose"],
            "compact_ok": per_compact[name]["allclose"],
        }
    ratios = [v["ratio"] for v in excess.values() if v["ratio"] is not None]
    excess_positive = sum(1 for v in excess.values() if v["excess"] > 0.0)
    result["diagnostic"] = {
        "common_params": len(common),
        "intrinsic_failures": result["intrinsic_noise"]["failure_count"],
        "compact_failures": result["compact_effect"]["failure_count"],
        "excess_positive_params": excess_positive,
        "ratio_median": float(sorted(ratios)[len(ratios) // 2]) if ratios else None,
        "ratio_max": float(max(ratios)) if ratios else None,
        "worst_excess10": sorted(
            [{"name": k, **v} for k, v in excess.items()], key=lambda d: -d["excess"]
        )[:10],
    }

    local_pass = (
        result["plan"]["members_eq"]
        and result["dcp_load"]["missing_count"] == 0
        and result["dcp_load"]["unexpected_count"] == 0
        and result["compact_effect"]["loss_eq"]
        and result["compact_effect"]["failure_count"] <= result["intrinsic_noise"]["failure_count"]
        and excess_positive <= max(1, len(common) // 2)
        and (result["diagnostic"]["ratio_max"] is None or result["diagnostic"]["ratio_max"] <= 2.0)
    )
    result["local_pass"] = bool(local_pass)
    result["raw_allclose_pass"] = bool(
        result["plan"]["members_eq"]
        and result["dcp_load"]["missing_count"] == 0
        and result["dcp_load"]["unexpected_count"] == 0
        and result["compact_effect"]["loss_eq"]
        and result["compact_effect"]["failure_count"] == 0
    )

    all_ok = [None] * world_size
    dist.all_gather_object(all_ok, bool(local_pass))
    result["all_ranks_pass"] = bool(all(all_ok))
    if all(all_ok):
        result["verdict"] = "PASS_CANDIDATE"
    else:
        result["verdict"] = "FAIL_PARITY"
    result["finished_unix"] = time.time()
    _emit(result, rank)

def math_isclose(a, b, rel_tol, abs_tol):
    return abs(a - b) <= max(rel_tol * max(abs(a), abs(b)), abs_tol)

def _emit(result, rank):
    out = Path("/tmp/v3_gpu_parity_rank%d.json" % rank)
    out.write_text(json.dumps(result, indent=2, sort_keys=True, default=str))
    print(f"[rank{rank}] verdict={result.get('verdict')} local_pass={result.get('local_pass')} -> {out}", flush=True)

if **name** == "**main**":
    main()
