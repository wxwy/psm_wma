"""Controlled old-production regression and native identity baseline; CPU only."""
from pathlib import Path
import hashlib
import json
import os
import subprocess
import sys
import xml.etree.ElementTree as ET

root = Path('/workspace/work/psm-wma-stage3a-h1-formal')
child = root / 'cosmos-framework'
out = Path('/workspace/work/issue24-stage3a-h1-evidence/red-old-code')
out.mkdir(exist_ok=False)
env = os.environ.copy()
overrides = dict(CUDA_VISIBLE_DEVICES='', PYTEST_DISABLE_PLUGIN_AUTOLOAD='1', PYTEST_ADDOPTS='',
                 HF_HUB_OFFLINE='1', TRANSFORMERS_OFFLINE='1', PYTHONDONTWRITEBYTECODE='1',
                 OMP_NUM_THREADS='1', MKL_NUM_THREADS='1', OPENBLAS_NUM_THREADS='1',
                 PYTHONPATH=os.pathsep.join(map(str, (root, child, root/'scripts'))),
                 RUFF_CACHE_DIR=str(out/'cache/ruff'), XDG_CACHE_HOME=str(out/'cache/xdg'))
env.update(overrides)
(out/'environment_overrides.json').write_text(json.dumps(overrides,indent=2)+'\n')
tests = [child/'cosmos_framework/inference/robocasa_vae_daemon_test.py',
         child/'cosmos_framework/scripts/robocasa_feature_http_test.py',
         root/'tools/v3/eval_v2_batch_tests/shared_vae_test.py']
commands=[]
def run(name, args):
    with (out/(name+'.log')).open('w') as log:
        result=subprocess.run(args,cwd=root,env=env,stdout=log,stderr=subprocess.STDOUT,timeout=120)
    commands.append(dict(name=name,argv=list(map(str,args)),cwd=str(root),exit_code=result.returncode))
    return result.returncode
assert run('ruff_fix',[sys.executable,'-m','ruff','check','--fix',*map(str,tests)])==0
assert run('ruff_format',[sys.executable,'-m','ruff','format',*map(str,tests)])==0
production=child/'cosmos_framework/inference/robocasa_vae_daemon.py'
old=subprocess.check_output(['git','-C',str(child),'show','6842b329f71fff8b9813b2d00b83edfc61e3e505:cosmos_framework/inference/robocasa_vae_daemon.py'])
assert production.read_bytes()==old, 'RED must run the unmodified old production source'
before=hashlib.sha256(old).hexdigest()
probe='''import json
from cosmos_framework.inference.robocasa_vae_daemon_test import _tokenizer
from cosmos_framework.inference.robocasa_vae_daemon import worker_constructor_config, loaded_vae_identity
from cosmos_framework.inference.robocasa_vae_protocol import canonical_json
t=_tokenizer()
print(json.dumps({'constructor':worker_constructor_config(t),'canonical_constructor':canonical_json(worker_constructor_config(t)).decode(),'identity':loaded_vae_identity(t)},sort_keys=True))
'''
assert run('native_baseline',[sys.executable,'-c',probe])==0
code=run('pytest',[sys.executable,'-m','pytest','--noconftest','-c','/dev/null','-p','no:cacheprovider','-q',
    str(tests[0])+'::test_real_omegaconf_constructor_equals_native_without_mutating_order_or_values',
    str(tests[0])+'::test_real_omegaconf_loaded_identity_matches_native_and_preserves_weight_authority',
    '--basetemp='+str(out/'tmp'),'--junitxml='+str(out/'junit.xml')])
suite=next(ET.parse(out/'junit.xml').getroot().iter('testsuite'))
log=(out/'pytest.log').read_text()
assert code==1 and suite.attrib['failures']=='2' and suite.attrib['errors']=='0'
assert 'ListConfig' in log and 'invalid canonical JSON' in log
assert hashlib.sha256(production.read_bytes()).hexdigest()==before
(out/'result.json').write_text(json.dumps(dict(status='RED_EXPECTED',pytest_exit=code,
    old_child='6842b329f71fff8b9813b2d00b83edfc61e3e505',production_sha256_before=before,
    production_sha256_after=before,tests_injected=True,source_note='Old production byte-identical; new tests injected into isolated worktree; not a clean Formal Pair gate.',
    test_sha256={str(p.relative_to(root)):hashlib.sha256(p.read_bytes()).hexdigest() for p in tests},
    failures=2,errors=0,skipped=0,commands=commands),indent=2)+'\n')
print(json.dumps(dict(status='RED_EXPECTED',pytest_exit=code,failures=2,production_unchanged=True,output=str(out))))
